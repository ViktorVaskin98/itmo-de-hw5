kind: Job

  name: ${NAME}
  namespace: ${NAMESPACE}
  labels:
    app.kubernetes.io/name: hhlake
    hw5/run: ${NAME}
    hw5/stage: hw3-job
spec:
  backoffLimit: 2
  podFailurePolicy:
    rules:
      - action: FailJob
        onExitCodes:
          containerName: mapreduce
          operator: In
          values: [137]
  template:
    metadata:
      labels:
        app.kubernetes.io/name: hhlake
        hw5/run: ${NAME}
        hw5/stage: hw3-job
    spec:
      restartPolicy: Never
      priorityClassName: ${PRIORITY}
      nodeSelector:
        hw5/pool: batch
      containers:
        - name: mapreduce
          image: hw5-hhlake:0.3.0
          imagePullPolicy: Never
          command:
            - python
            - -m
            - hhlake.kube.memwatch
            - --
            - python
            - -m
            - hhlake.mapreduce
            - --workers
            - "${WORKERS}"
            - --workdir
            - /work
            - --output
            - results/k8s/${NAME}
          env:
            - name: S3_ENDPOINT
              value: ${S3_ENDPOINT}
          envFrom:
            - secretRef:
                name: s3-credentials
          resources:
            requests:
              cpu: "${CPU_REQUEST}"
              memory: ${MEMORY}
            limits:
              cpu: "${CPU_LIMIT}"
              memory: ${MEMORY}
          volumeMounts:
            - name: work
              mountPath: /work
      volumes:
        - name: work
          emptyDir: {}
